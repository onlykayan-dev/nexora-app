import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; const LIMIT = 1000; const WINDOW_MS = 8 * 60 * 60 * 1000; // 8 hours Deno.serve(async (req) => { // CORS if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders, }); } if (req.method !== "POST") { return new Response( JSON.stringify({ success: false, error: "Only POST requests are allowed", }), { status: 405, headers: { ...corsHeaders, "Content-Type": "application/json", }, }, ); } try { const supabaseUrl = Deno.env.get("SUPABASE_URL"); const supabaseKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); if (!supabaseUrl || !supabaseKey) { throw new Error("Supabase environment variables are missing"); } const supabase = createClient( supabaseUrl, supabaseKey, { auth: { autoRefreshToken: false, persistSession: false, }, }, ); // شناسه کاربر const authHeader = req.headers.get("Authorization"); let identifier = ""; if (authHeader) { const token = authHeader.replace("Bearer ", ""); const { data: { user }, } = await supabase.auth.getUser(token); if (user?.id) { identifier = `user:${user.id}`; } } // اگر کاربر لاگین نبود، IP استفاده می‌شود if (!identifier) { const forwarded = req.headers.get("x-forwarded-for"); const realIp = req.headers.get("x-real-ip"); const ip = forwarded?.split(",")[0]?.trim() || realIp || "unknown"; identifier = `ip:${ip}`; } const now = Date.now(); // جدول rate_limits const { data: existing, error: readError } = await supabase .from("rate_limits") .select("*") .eq("identifier", identifier) .maybeSingle(); if (readError) { throw readError; } let count = 0; let startedAt = now; if (existing) { count = Number(existing.count || 0); startedAt = new Date(existing.started_at).getTime(); // اگر 8 ساعت تمام شده، سهمیه از نو شروع می‌شود if (now - startedAt >= WINDOW_MS) { count = 0; startedAt = now; } } // اگر سهمیه تمام شده if (count >= LIMIT) { const resetAt = startedAt + WINDOW_MS; const remainingMs = Math.max(0, resetAt - now); const remainingSeconds = Math.ceil( remainingMs / 1000, ); return new Response( JSON.stringify({ success: false, error: "LIMIT_REACHED", message: "سهمیه 1000 درخواست شما تمام شده است.", limit: LIMIT, used: count, remaining: 0, resetAt: new Date(resetAt).toISOString(), remainingSeconds, }), { status: 429, headers: { ...corsHeaders, "Content-Type": "application/json", "Retry-After": String(remainingSeconds), }, }, ); } // یک درخواست مصرف شد count++; const { error: upsertError } = await supabase .from("rate_limits") .upsert( { identifier, count, started_at: new Date(startedAt).toISOString(), updated_at: new Date(now).toISOString(), }, { onConflict: "identifier", }, ); if (upsertError) { throw upsertError; } const resetAt = startedAt + WINDOW_MS; const remainingMs = Math.max(0, resetAt - now); const remainingSeconds = Math.ceil( remainingMs / 1000, ); return new Response( JSON.stringify({ success: true, message: "درخواست با موفقیت ثبت شد.", limit: LIMIT, used: count, remaining: LIMIT - count, resetAt: new Date(resetAt).toISOString(), remainingSeconds, windowHours: 8, }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json", }, }, ); } catch (error) { console.error(error); return new Response( JSON.stringify({ success: false, error: "SERVER_ERROR", message: "خطایی در سرور رخ داد.", }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json", }, }, ); } });
